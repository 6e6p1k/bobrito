NVCC = nvcc
CXX = g++
FLAGS = -std=c++17
FLAGS += -I$(CONDA_PREFIX)/include
NVCC_LDFLAGS = -L$(CONDA_PREFIX)/lib -lcublas -lcudart

CUDA_OBJS = ops.o allocator.o kernels.o
CPU_OBJS = utils.o graph_viz.o visualize_graph.o backward.o node.o

# Ensure CONDA_PREFIX is recognized
LDFLAGS += -L$(CONDA_PREFIX)/lib -lcublas -lcudart

all: visualize_graph

visualize_graph: $(CUDA_OBJS) $(CPU_OBJS) 
	$(NVCC) $(FLAGS) $^ -o $@ $(NVCC_LDFLAGS)

%.o: %.cu
	$(NVCC) $(FLAGS) -c $< -o $@

%.o: %.cpp
	$(CXX) $(FLAGS) -c $< -o $@

clean:
	rm -f *.o visualize_graph
